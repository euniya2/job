결과물 보기 ↗ 
[역할 부여]
너는 참여형 웹 도구 전문 개발자이자 UI/UX 디자이너야. 모임이나 워크숍 진행자가 현장에서 바로 쓸 수 있는 실용적이고 보기 좋은 웹 앱을 만들어줘.
[작업 목표]
참가자 중 한 명을 공정하게 뽑는 '오늘의 주인공' 웹 앱을 단일 HTML 파일로 제작해줘.
[핵심 기능]
명단 관리: 엑셀이나 메모장에서 복사한 여러 이름을 한꺼번에 붙여넣어 등록할 수 있어야 한다(줄바꿈과 쉼표 모두 구분자로 인식할 것)
중복 방지: '한 번 뽑힌 사람은 다시 안 나오게' 하는 체크박스를 제공한다. 이미 뽑힌 사람은 명단에서 흐리게 표시한다.
셔플 연출: 뽑기 버튼을 누르면 이름이 빠르게 돌아가다가 점점 느려지면서 멈춘다. 2초 정도 끌어서 긴장감을 준다.
진행 상황: '3 / 20 완료' 처럼 지금까지 몇 명이 뽑혔는지 보여준다. 전원이 다 뽑히면 뽑기 버튼이 비활성화된다.
기록 초기화: 뽑힌 기록을 지우고 처음부터 다시 시작하는 버튼.
[디자인 및 기술 제약]
디자인 컨셉: 밝은 종이 무대 위로 부드러운 조명이 내려앉는 느낌, 라이트 세이지 그린 계열에 연한 올리브 색상과 카라멜 브라운 계열 포인트
당첨된 이름은 화면 한가운데에 아주 크게.
가독성: 빔프로젝터로 띄웠을 때 교실 맨 뒷자리에서도 이름이 읽혀야 한다.
  당첨자 이름은 화면 높이의 10% 이상 크기로 할 것.
레이아웃: 왼쪽에 명단 관리, 오른쪽에 무대. 화면이 좁아지면 위아래로 쌓인다.
기술: 외부 라이브러리는 CDN으로만. HTML, CSS, JS를 하나의 파일에 통합할 것.

/**
 * 신청 데이터 저장 함수
 * - appendRow를 사용하여 동시 제출 시 데이터 덮어쓰기 방지
 * - 시트명: '접수'
 * - 헤더 구성: 신청일 · 신청 항목 · 이름 · 상태 · 비고
 * 
 * @param {Object} data { name: string, category: string, note: string }
 * @return {Object} 처리 결과
 */
function submitApplication(data) {
  try {
    if (!data || !data.name || !data.category) {
      throw new Error('이름과 신청 항목은 필수 입력 항목입니다.');
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('접수');

    // '접수' 시트가 없을 경우 자동 생성 및 헤더 초기화
    if (!sheet) {
      sheet = ss.insertSheet('접수');
      sheet.appendRow(['신청일', '신청 항목', '이름', '상태', '비고']);
    }

    // 서버 시간 기준 신청일 자동 생성 (형식: YYYY-MM-DD HH:mm:ss)
    var timestamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'Asia/Seoul', 'yyyy-MM-dd HH:mm:ss');
    var status = '접수';
    var note = data.note || '';

    // 동시성 안전 저장을 위해 appendRow만 사용
    sheet.appendRow([
      timestamp,
      data.category,
      data.name,
      status,
      note
    ]);

    return {
      success: true,
      message: '접수가 성공적으로 완료되었습니다.'
    };
  } catch (err) {
    throw new Error(err.message || '데이터 저장 중 알 수 없는 오류가 발생했습니다.');
  }
}

/**
 * 접수 내역 및 건수 조회 함수
 * 
 * @return {Object} { totalCount: number, list: Array }
 */
function getApplications() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('접수');

    if (!sheet) {
      return { totalCount: 0, list: [] };
    }

    var values = sheet.getDataRange().getValues();
    
    // 데이터가 헤더(1행) 이하인 경우
    if (values.length <= 1) {
      return { totalCount: 0, list: [] };
    }

    // 헤더 제외한 데이터 행 추출
    var rows = values.slice(1);
    var totalCount = rows.length;

    var list = rows.map(function(row) {
      var dateVal = row[0];
      var formattedDate = '';
      if (dateVal instanceof Date) {
        formattedDate = Utilities.formatDate(dateVal, Session.getScriptTimeZone() || 'Asia/Seoul', 'yyyy-MM-dd HH:mm');
      } else {
        formattedDate = String(dateVal);
      }

      return {
        date: formattedDate,
        category: String(row[1] || ''),
        name: String(row[2] || ''),
        status: String(row[3] || ''),
        note: String(row[4] || '')
      };
    });

    // 최신 신청 건이 위로 오도록 역순 정렬
    list.reverse();

    return {
      totalCount: totalCount,
      list: list
    };
  } catch (err) {
    throw new Error('목록 조회 실패: ' + (err.message || '알 수 없는 오류'));
  }
}

        category: '원데이 워크숍',
        name: '홍길동',
        status: '접수',
        note: '온라인 참여 링크 요청'
      },
      {
        date: '2026-10-20 10:15',
        category: '정기 교육',
        name: '김철수',
        status: '접수',
        note: ''
      }
    ];

    // 화면 안내 메시지 표시 (성공 / 실패 원문 그대로 노출)
    function showBanner(type, message) {
      banner.className = 'status-banner ' + type;
      banner.textContent = message;
    }

    function clearBanner() {
      banner.className = 'status-banner';
      banner.textContent = '';
    }

    // 목록 렌더링 함수
    function renderList(data) {
      totalBadge.textContent = (data.totalCount || 0) + '건';

      if (!data.list || data.list.length === 0) {
        listContainer.innerHTML = '<div class="empty-state">아직 접수된 내역이 없습니다.</div>';
        return;
      }

      var html = '';
      for (var i = 0; i < data.list.length; i++) {
        var item = data.list[i];
        html += '<div class="item-card">';
        html += '  <div class="item-top">';
        html += '    <span class="item-name">' + escapeHtml(item.name) + '</span>';
        html += '    <span class="item-status">' + escapeHtml(item.status) + '</span>';
        html += '  </div>';
        html += '  <div class="item-cat">' + escapeHtml(item.category) + '</div>';
        html += '  <div class="item-meta">신청일시: ' + escapeHtml(item.date) + '</div>';
        if (item.note && item.note.trim() !== '') {
          html += '  <div class="item-note">' + escapeHtml(item.note) + '</div>';
        }
        html += '</div>';
      }
      listContainer.innerHTML = html;
    }

    // HTML 특수문자 치환 (XSS 방지)
    function escapeHtml(text) {
      if (!text) return '';
      return String(text)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
    }

    // 목록 불러오기
    function loadApplications() {
      refreshBtn.disabled = true;
      refreshBtn.textContent = '조회중...';

      // Google Apps Script 환경일 때
      if (isGasEnvironment) {
        google.script.run
          .withSuccessHandler(function(result) {
            refreshBtn.disabled = false;
            refreshBtn.textContent = '새로고침';
            renderList(result);
          })
          .withFailureHandler(function(error) {
            refreshBtn.disabled = false;
            refreshBtn.textContent = '새로고침';
            showBanner('error', '목록 새로고침 오류: ' + (error.message || error));
          })
          .getApplications();
      } else {
        // 프리뷰 / 로컬 테스트 환경 지원 (ReferenceError 방지)
        setTimeout(function() {
          refreshBtn.disabled = false;
          refreshBtn.textContent = '새로고침';
          renderList({
            totalCount: mockDataList.length,
            list: mockDataList
          });
        }, 300);
      }
    }

    // 폼 제출 이벤트 핸들러
    form.addEventListener('submit', function(e) {
      e.preventDefault();
      clearBanner();

      var nameVal = document.getElementById('applicantName').value.trim();
      var categoryVal = document.getElementById('categorySelect').value;
      var noteVal = document.getElementById('applicantNote').value.trim();

      if (!nameVal || !categoryVal) {
        showBanner('error', '이름과 신청 항목을 모두 입력해 주세요.');
        return;
      }

      var payload = {
        name: nameVal,
        category: categoryVal,
        note: noteVal
      };

      submitBtn.disabled = true;
      submitBtn.textContent = '접수 처리 중...';

      // Google Apps Script 환경일 때
      if (isGasEnvironment) {
        google.script.run
          .withSuccessHandler(function(response) {
            submitBtn.disabled = false;
            submitBtn.textContent = '신청서 제출';
            showBanner('success', response.message || '접수가 완료되었습니다.');
            form.reset();
            loadApplications();
          })
          .withFailureHandler(function(error) {
            submitBtn.disabled = false;
            submitBtn.textContent = '신청서 제출';
            showBanner('error', '접수 실패: ' + (error.message || error));
          })
          .submitApplication(payload);
      } else {
        // 프리뷰 / 로컬 테스트 환경 지원
        setTimeout(function() {
          submitBtn.disabled = false;
          submitBtn.textContent = '신청서 제출';
          
          var now = new Date();
          var dateStr = now.getFullYear() + '-' +
            String(now.getMonth() + 1).padStart(2, '0') + '-' +
            String(now.getDate()).padStart(2, '0') + ' ' +
            String(now.getHours()).padStart(2, '0') + ':' +
            String(now.getMinutes()).padStart(2, '0');

          mockDataList.unshift({
            date: dateStr,
            category: payload.category,
            name: payload.name,
            status: '접수',
            note: payload.note
          });

          showBanner('success', '[미리보기 모드] 접수가 성공적으로 처리되었습니다.');
          form.reset();
          loadApplications();
        }, 500);
      }
    });

    // 새로고침 버튼 클릭 시
    refreshBtn.addEventListener('click', function() {
      clearBanner();
      loadApplications();
    });

    // 화면 로드 시 자동 최초 1회 목록 조회
    window.addEventListener('DOMContentLoaded', function() {
      loadApplications();
    });
  </script>

A
관리자만 전체 보기

    

      
프롬프트 ⑧

      
접수 현황 목록은 신청자에게 보이지 않게 숨겨 줘.
접수가 끝나면 방금 신청한 내용만 확인 카드로 보여 줘.

화면 아래에 '관리자 보기' 버튼을 만들고, 암호를 입력해서 맞으면
그때 전체 접수 목록과 건수가 보이게 해 줘.
암호 확인은 Code.gs에서 하고, 암호는 Code.gs 맨 위에 상수로 모아 줘.
암호가 틀리면 화면에 틀렸다고 표시해 줘.

      
    

    
암호는 가림막이지 보안은 아닙니다. 개인정보를 받는 접수라면 구글 로그인으로 제한해야 합니다.

  


  

    
B
접수 내용 수정·삭제하기

    

      
프롬프트 ⑨

      
접수 목록에서 각 건을 수정하거나 삭제할 수 있게 해 줘.
목록의 줄마다 수정·삭제 버튼을 두고,
수정은 이름 · 신청 항목 · 요청사항 · 상태를 바꿀 수 있게 해 줘.
시트에서도 해당 줄이 같이 바뀌고 지워지게 해 줘.

어느 줄인지 구분할 수 있도록 저장할 때 각 건에 고유 번호를 함께 기록해 줘.
수정과 삭제는 관리자 보기에서만 되게 해 줘.

      
    

    
고유 번호가 없으면 엉뚱한 줄이 수정·삭제됩니다. 이 줄은 빼지 마세요.

프롬프트 ⑩

      
신청 화면에 이메일 입력 칸을 추가해 줘. 필수 입력으로 해 줘.
시트 첫 줄은 신청일 · 신청 항목 · 이름 · 이메일 · 상태 · 비고
여섯 칸으로 바꿔 줘.

접수가 저장된 뒤에 신청자 본인에게 접수 확인 메일을 보내 줘.
제목은 '[접수 완료] ○○ 신청이 접수되었습니다' 형태로 하고,
본문에는 이름 · 신청 항목 · 신청일 · 요청사항이 들어가게 해 줘.

메일 발송이 실패해도 접수는 시트에 저장되게 해 주고,
실패하면 화면에 알려 줘.
